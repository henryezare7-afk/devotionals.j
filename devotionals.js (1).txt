// THE MAGIC - picks devotional based on today's date
export function getTodayDevotional() {
  const today = new Date();
  const dayOfMonth = today.getDate(); // 1 to 31
  const index = (dayOfMonth - 1) % devotionals.length;
  return devotionals[index];
}

// Also get Verse of the Day - same as devotional verse
export function getVerseOfDay() {
  const d = getTodayDevotional();
  return { verse: d.verse, ref: d.ref };
}
